func polygon_perimeter(polygon poly):
    result = 0
    n = len(poly.points)
    
    for i in range(0, n):
        p1 = poly.points[i]
        p2 = poly.points[(i + 1) % n]
        result = result + distance(p1, p2)
    return result

func triangle_area(num a, num b, num c):
    func heron_formula(num sem_perim, num x, num y, num z):
        result = sem_perim*(sem_perim - x)*(sem_perim - y)*(sem_perim - z)
        return result

    s = (a + b + c)/2
    area = heron_formula(s, a, b, c)
    
    if area > 0:
        return sqrt(area)
    else:
        return 0

func triangle_type(num a, num b, num c):
    if a == b and b == c:
        return 1 
    else:
        return 2
